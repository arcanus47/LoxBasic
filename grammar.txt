program     : NEWLINE* statements? NEWLINE* EOF

statements  : statement (NEWLINE+ statement)*

statement   : KEYWORD:RETURN expr?
            : KEYWORD:CONTINUE
            : KEYWORD:BREAK
            : expr

expr        : KEYWORD:VAR IDENTIFIER (EQ expr)?
            : assignment-expr

assignment-expr
            : logical-expr ((EQ|PLUSEQ|MINUSEQ|MULEQ|DIVEQ|MODEQ) assignment-expr)?
            
logical-expr
            : comp-expr ((KEYWORD:AND|KEYWORD:OR) comp-expr)*

comp-expr   : (KEYWORD:NOT|NOT) comp-expr
            : arith-expr ((EE|NE|LT|GT|LTE|GTE) arith-expr)*

arith-expr  : term ((PLUS|MINUS) term)*

term        : factor ((MUL|DIV|MOD) factor)*

factor      : (PLUS|MINUS) factor
            : power

power       : call ((POW) factor)?

call        : atom
              (LPAREN (expr (COMMA expr)*)? COMMA? RPAREN
              | LSQUARE expr RSQUARE)*

atom        : INT
            : FLOAT
            : STRING
            : IDENTIFIER
            : KEYWORD:TRUE
            : KEYWORD:FALSE
            : KEYWORD:NULL
            : LPAREN expr RPAREN
            : list-expr
            : if-expr
            : for-expr
            : while-expr
            : func-def

list-expr   : LSQUARE
              (expr (COMMA expr)* COMMA?)?
              RSQUARE

if-expr     : KEYWORD:IF expr KEYWORD:THEN
              if-body
              if-branches?
              KEYWORD:END?

if-body     : statement
            : NEWLINE statements NEWLINE*

if-branches : KEYWORD:ELIF expr KEYWORD:THEN
              if-body
              if-branches?
            : KEYWORD:ELSE
              else-body

else-body   : statement
            : NEWLINE statements NEWLINE*

for-expr    : KEYWORD:FOR IDENTIFIER EQ expr
              KEYWORD:TO expr
              (KEYWORD:STEP expr)?
              KEYWORD:THEN
              for-body
              KEYWORD:END?

            : KEYWORD:FOR IDENTIFIER KEYWORD:IN expr
              KEYWORD:THEN
              for-body
              KEYWORD:END?

for-body    : statement
            : NEWLINE statements NEWLINE*

while-expr  : KEYWORD:WHILE expr KEYWORD:THEN
              while-body
              KEYWORD:END?

while-body  : statement
            : NEWLINE statements NEWLINE*

func-def    : KEYWORD:FUN IDENTIFIER?
              LPAREN
              (IDENTIFIER (COMMA IDENTIFIER)* COMMA?)?
              RPAREN
              ARROW expr

            : KEYWORD:FUN IDENTIFIER?
              LPAREN
              (IDENTIFIER (COMMA IDENTIFIER)* COMMA?)?
              RPAREN
              NEWLINE
              statements
              NEWLINE*
              KEYWORD:END
